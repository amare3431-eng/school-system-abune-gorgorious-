            { id: "ST/003", name: "ዮናስ ታደሰ", class: "12ኛ ክፍል B", math: 45, english: 55, amharic: 60 },
            { id: "ST/004", name: "ሰላም አሰፋ", class: "12ኛ ክፍል C", math: 78, english: 82, amharic: 85 }

        // Initialize application on load

        // Render Table data
        function renderTable(data) {
            const tbody = document.getElementById('studentTableBody');
            const emptyState = document.getElementById('emptyState');
            tbody.innerHTML = '';

            if (data.length === 0) {
                emptyState.classList.remove('hidden');
                return;
            } else {
                emptyState.classList.add('hidden');
            }

            data.forEach((student, index) => {
                const avg = Math.round((student.math + student.english + student.amharic) / 3);
                const isPass = avg >= 50;

                const tr = document.createElement('tr');
                tr.className = "hover:bg-gray-50 transition border-b border-gray-100";
                tr.innerHTML = `
                    <td class="py-3 px-4 font-medium text-gray-800">${student.name}</td>
                    <td class="py-3 px-4 text-gray-500 text-xs">${student.id}</td>
                    <td class="py-3 px-4">${student.class}</td>
                    <td class="py-3 px-4">${student.math}</td>
                    <td class="py-3 px-4">${student.english}</td>
                    <td class="py-3 px-4">${student.amharic}</td>
                    <td class="py-3 px-4 font-semibold">${avg}%</td>
                    <td class="py-3 px-4">
                        <span class="px-2 py-1 rounded-full text-xs font-semibold ${isPass ? 'bg-emerald-50 text-emerald-600' : 'bg-rose-50 text-rose-600'}">
                            ${isPass ? 'አልፏል' : 'ወድቋል'}
                        </span>
                    </td>
                    <td class="py-3 px-4 text-center space-x-2 space-x-reverse">
                        <button onclick="viewReport(${index})" class="text-indigo-600 hover:text-indigo-800 p-1" title="ሪፖርት አሳይ"><i class="fa-solid fa-eye"></i></button>
                        <button onclick="editStudent(${index})" class="text-amber-600 hover:text-amber-800 p-1" title="አስተካክል"><i class="fa-solid fa-pen-to-square"></i></button>
                        <button onclick="deleteStudent(${index})" class="text-rose-600 hover:text-rose-800 p-1" title="ሰርዝ"><i class="fa-solid fa-trash"></i></button>
                    </td>
                `;
                tbody.appendChild(tr);
            });
        }

        // Update Dashboard Statistics
        function updateDashboardStats() {
            document.getElementById('statTotalStudents').innerText = students.length;

            if (students.length === 0) {
                document.getElementById('statAvgScore').innerText = '0.0';
                document.getElementById('statPassRate').innerText = '0%';
                return;
            }

            let totalSum = 0;
            let passCount = 0;

            students.forEach(s => {
                const avg = (s.math + s.english + s.amharic) / 3;
                totalSum += avg;
                if (avg >= 50) passCount++;
            });

            const overallAvg = (totalSum / students.length).toFixed(1);
            const passRate = Math.round((passCount / students.length) * 100);

            document.getElementById('statAvgScore').innerText = overallAvg + '%';
            document.getElementById('statPassRate').innerText = passRate + '%';
        }

        // Filter Students
        function filterStudents() {
            const query = document.getElementById('searchInput').value.toLowerCase();
            const classVal = document.getElementById('classFilter').value;
            const statusVal = document.getElementById('statusFilter').value;

            const filtered = students.filter(s => {
                const matchesQuery = s.name.toLowerCase().includes(query) || s.id.toLowerCase().includes(query);
                const matchesClass = classVal === "" || s.class === classVal;
                
                const avg = (s.math + s.english + s.amharic) / 3;
                const isPass = avg >= 50;
                let matchesStatus = true;
                if (statusVal === 'pass') matchesStatus = isPass;
                if (statusVal === 'fail') matchesStatus = !isPass;

                return matchesQuery && matchesClass && matchesStatus;
            });

            renderTable(filtered);
        }

        // Modal Controls
        function openAddModal() {
            document.getElementById('modalTitle').innerText = "አዲስ ተማሪ መዝግብ";

            document.getElementById('studentIndex').value = "";
            document.getElementById('studentModal').classList.remove('hidden');
        }

        function closeModal() {
            document.getElementById('studentModal').classList.add('hidden');
        }

        function editStudent(index) {
            const s = students[index];
            document.getElementById('modalTitle').innerText = "የተማሪ መረጃ ማስተካከያ";
            document.getElementById('studentIndex').value = index;
            document.getElementById('fullName').value = s.name;
            document.getElementById('studentId').value = s.id;
            document.getElementById('studentClass').value = s.class;
            document.getElementById('mathScore').value = s.math;
            document.getElementById('englishScore').value = s.english;
            document.getElementById('amharicScore').value = s.amharic;
            document.getElementById('studentModal').classList.remove('hidden');
        }

        // Save Student (Add or Update)

            event.preventDefault();
            const index = document.getElementById('studentIndex').value;
            const studentData = {
                name: document.getElementById('fullName').value,
                id: document.getElementById('studentId').value,
                class: document.getElementById('studentClass').value,
                math: Number(document.getElementById('mathScore').value),
                english: Number(document.getElementById('englishScore').value),
                amharic: Number(document.getElementById('amharicScore').value)
            };

            if (index === "") {
                students.push(studentData);
            } else {
                students[index] = studentData;
            }

            localStorage.setItem('students_data', JSON.stringify(students));
            closeModal();
            renderTable(students);
            updateDashboardStats();
            updateCharts();
        }

        // Delete Student
        function deleteStudent(index) {
            if (confirm("ይህንን ተማሪ መረጃ መሰረዝ ይፈልጋሉ?")) {
                students.splice(index, 1);
                localStorage.setItem('students_data', JSON.stringify(students));
                renderTable(students);
                updateDashboardStats();
                updateCharts();
            }
        }

        // Report Card Modal View
        function viewReport(index) {
            const s = students[index];
            const avg = Math.round((s.math + s.english + s.amharic) / 3);
            const isPass = avg >= 50;

            document.getElementById('repName').innerText = s.name;
            document.getElementById('repId').innerText = s.id;
            document.getElementById('repClass').innerText = s.class;
            document.getElementById('repMath').innerText = s.math + " / 100";
            document.getElementById('repEnglish').innerText = s.english + " / 100";
            document.getElementById('repAmharic').innerText = s.amharic + " / 100";
            document.getElementById('repTotal').innerText = avg + "%";
            
            const statusEl = document.getElementById('repStatus');
            statusEl.innerText = isPass ? 'አልፏል (Passed)' : 'ወድቋል (Failed)';
            statusEl.className = `px-2.5 py-1 rounded-full text-xs font-semibold ${isPass ? 'bg-emerald-50 text-emerald-600' : 'bg-rose-50 text-rose-600'}`;

            document.getElementById('reportModal').classList.remove('hidden');
        }

        function closeReportModal() {
            document.getElementById('reportModal').classList.add('hidden');
        }

        // Export data to CSV
        function exportToCSV() {
            let csvContent = "data:text/csv;charset=utf-8,\uFEFFስም,መለያ,ክፍል,ሒሳብ,እንግሊዝኛ,አማርኛ,አማካይ\n";
            students.forEach(s => {
                const avg = Math.round((s.math + s.english + s.amharic) / 3);
                csvContent += `"${s.name}","${s.id}","${s.class}",${s.math},${s.english},${s.amharic},${avg}\n`;
            });
            const encodedUri = encodeURI(csvContent);
            const link = document.createElement("a");
            link.setAttribute("href", encodedUri);
            link.setAttribute("download", "students_report.csv");
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }

        // Charts Initialization
        function initCharts() {
            // Score Chart
            const ctxScore = document.getElementById('scoreChart').getContext('2d');
            scoreChartInstance = new Chart(ctxScore, {
                type: 'doughnut',
                data: {
                    labels: ['እጅግ በጣም ጥሩ (85%+)', 'ጥሩ (70%-84%)', 'አማካይ (50%-69%)', 'አልፋሽ (<50%)'],
                    datasets: [{
                        data: getScoreDistribution(),
                        backgroundColor: ['#4f46e5', '#38bdf8', '#fbbf24', '#f43f5e']
                    }]
                },
                options: { responsive: true, maintainAspectRatio: false }
            });

            // Class Distribution Chart
            const ctxClass = document.getElementById('classChart').getContext('2d');
            classChartInstance = new Chart(ctxClass, {
                type: 'bar',
                data: {
                    labels: ['12ኛ ክፍል A', '12ኛ ክፍል B', '12ኛ ክፍል C'],
                    datasets: [{
                        label: 'የተማሪዎች ብዛት',
                        data: getClassDistribution(),
                        backgroundColor: '#4f46e5'
                    }]
                },
                options: { responsive: true, maintainAspectRatio: false }
            });
        }

        function getScoreDistribution() {
            let excel = 0, good = 0, avg = 0, fail = 0;
            students.forEach(s => {
                const a = (s.math + s.english + s.amharic) / 3;
                if (a >= 85) excel++;
                else if (a >= 70) good++;
                else if (a >= 50) avg++;
                else fail++;
            });
            return [excel, good, avg, fail];
        }

        function getClassDistribution() {
            let a = 0, b = 0, c = 0;
            students.forEach(s => {
                if (s.class.includes('A')) a++;
                else if (s.class.includes('B')) b++;
                else if (s.class.includes('C')) c++;
            });
            return [a, b, c];
        }

        function updateCharts() {
            if (scoreChartInstance && classChartInstance) {
                scoreChartInstance.data.datasets[0].data = getScoreDistribution();
                scoreChartInstance.update();

                classChartInstance.data.datasets[0].data = getClassDistribution();
                classChartInstance.update();
            }
        }
